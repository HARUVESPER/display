  <script>
    // 💡 關鍵：請將下方的雙引號內容，替換成你新部署的 Google Apps Script API 網址 (/exec 結尾)
    const gasUrl = "https://script.google.com/macros/s/AKfycbxtv3IkvwhFNf6VmU_A8RDQEiLV8rnKjBxra9eLpfcxOR_XLrXMqTzhbnrsBy_95Ak/exec";
    
    // 從外部向你的 Google 後端發送資料請求
    fetch(gasUrl)
      .then(response => response.json())
      .then(data => {
        renderProducts(data);
      })
      .catch(error => {
        console.error("衣櫥資料連線失敗:", error);
        document.getElementById('gallery').innerHTML = '<div class="loading">目前可能出現錯誤，衣櫃連線失敗</div>';
      });
    
    function renderProducts(products) {
      var gallery = document.getElementById('gallery');
      gallery.innerHTML = '';
      
      if (!products || products.length === 0) {
        gallery.innerHTML = '<div class="loading">目前可能出現錯誤,衣櫃空空如也</div>';
        return;
      }
      
      products.forEach(function(prod, index) {
        var card = document.createElement('div');
        card.className = 'product-card';
        var slidesHtml = '';
        
        prod.images.forEach(function(imgUrl) {
          // 💡 加上 referrerpolicy="no-referrer" 確保手機能順利載入 Google 直連縮圖
          slidesHtml += `<div class="swiper-slide"><img src="${imgUrl}" referrerpolicy="no-referrer" loading="lazy"></div>`;
        });
        
        card.innerHTML = `
          <div class="swiper">
            <div class="swiper-wrapper">${slidesHtml}</div>
            <div class="swiper-button-next"></div><div class="swiper-button-prev"></div>
          </div>
          <div class="product-info">
            <div class="product-name">${prod.name}</div>
            <div class="product-price">${prod.price}</div>
          </div>
        `;
        gallery.appendChild(card);
        new Swiper(card.querySelector(".swiper"), { loop: true });
      });
    }
  </script>
